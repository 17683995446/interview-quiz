数据模型

核心类型定义

// 题目类型
interface Question {
  id: number
  cat: CategoryId           // 分类: ai_basic, ml, dl, nlp, cv, algorithm, network, os, database, python
  diff: Difficulty          // 难度: easy, medium, hard
  type: QuestionType        // 类型: choice, truefalse
  q: string                 // 题目文本
  opts: string[]            // 选项
  ans: number               // 正确答案索引
  exp: string               // 解析说明
}

// 答题记录
type Records = Record

// 收藏集合
type Bookmarks = Set

// 每日学习记录
type DailyRecords = Record

// 间隔重复卡片
interface SpacedRepetitionCard {
  questionId: number
  interval: number          // 复习间隔(天)
  repetitions: number       // 重复次数
  easeFactor: number        // 简易度因子(默认2.5)
  lastReviewDate: string    // ISO日期
  nextReviewDate: string    // ISO日期
}

// 刷题会话
interface QuizSession {
  questions: Question[]
  current: number           // 当前题号
  score: number             // 得分
  answered: boolean         // 是否已答
  results: Array<{ questionId: number; correct: boolean }>
  config: QuizConfig
}
